func EnsureMSSQLServerCRDs(crdClient crd_cs.Interface, removeUnusedCRDs bool) error {
  crds := MSSQLServerSelfCRDs
  if GitOpsEnabled {
    crds = append(crds, toSlice(gitopsapi.MSSQLServer{}.CustomResourceDefinition())...)
  }

  if features.DefaultFeatureGate.Enabled(features.MSSQLServer) {
    return apiextensions.RegisterWithOpts(crdClient, crds, preserveConversion)
  } else if removeUnusedCRDs {
    return safeDelete(crdClient, crds)
  }
  return nil
}
