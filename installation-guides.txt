        --version v2026.1.19 \
        --namespace kubedb --create-namespace \

        --set kubedb-provisioner.replicaCount=3 \
        --set-file global.license=$HOME/Downloads/neaj-hanadb-license.txt \

  "spec": {
    "template": {
      "spec": {
        "affinity": {
          "podAntiAffinity": {
            "preferredDuringSchedulingIgnoredDuringExecution": [
              {
                "weight": 100,
                "podAffinityTerm": {
                  "labelSelector": {
                    "matchLabels": {
                      "app.kubernetes.io/instance": "kubedb",
                      "app.kubernetes.io/name": "kubedb-provisioner"
                    }
                  },
                  "topologyKey": "kubernetes.io/hostname"
                }
              }
            ]
          }
        }
      }
    }
  }
